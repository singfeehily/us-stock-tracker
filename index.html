from http.server import BaseHTTPRequestHandler import urllib.request import json from urllib.parse import urlparse, parse_qs class handler(BaseHTTPRequestHandler): def do_GET(self): query_components = parse_qs(urlparse(self.path).query) symbols_str = query_components.get("symbols", [""])[0] interval = query_components.get("interval", [""])[0].lower() if not symbols_str: self.send_response(400) self.send_header('Cache-Control', 'no-store') self.send_header('Access-Control-Allow-Origin', '*') self.end_headers() self.wfile.write("請提供股票代號".encode('utf-8')) return results = {} symbols = [s.strip().upper() for s in symbols_str.split(',') if s.strip()] headers = {'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'} try: for sym in symbols: try: if not interval: url = f"https://query1.finance.yahoo.com/v8/finance/chart/{sym}?range=1d&interval=1m&includePrePost=true" req = urllib.request.Request(url, headers=headers) with urllib.request.urlopen(req) as response: data = json.loads(response.read().decode()) meta = data['chart']['result'][0]['meta'] reg_price = meta.get('regularMarketPrice', 0) post_price = meta.get('postMarketPrice') pre_price = meta.get('preMarketPrice') # 固定的基準價：昨天正規市場收盤價 (絕對不能變動) prev_close = meta.get('chartPreviousClose', meta.get('previousClose', reg_price)) # 判斷當前有效價格 (盤前 > 盤後 > 常規) if pre_price: final_price = pre_price elif post_price: final_price = post_price else: final_price = reg_price # 計算準確的漲跌幅 (現價 - 昨天收盤價) change = final_price - prev_close change_pct = (change / prev_close * 100) if prev_close > 0 else 0 results[sym] = { 'price': round(final_price, 2), 'change': round(change, 2), 'change_pct': round(change_pct, 2), 'volume': meta.get('regularMarketVolume', 0) } else: yf_interval, yf_range = "1d", "6mo" if interval == '15m': yf_interval, yf_range = "15m", "20d" elif interval == '4h': yf_interval, yf_range = "60m", "3mo" url = f"https://query1.finance.yahoo.com/v8/finance/chart/{sym}?range={yf_range}&interval={yf_interval}&includePrePost=true" req = urllib.request.Request(url, headers=headers) with urllib.request.urlopen(req) as response: data = json.loads(response.read().decode()) chart_result = data['chart']['result'][0] meta = chart_result['meta'] reg_price = meta.get('regularMarketPrice', 0) post_price = meta.get('postMarketPrice') pre_price = meta.get('preMarketPrice') prev_close = meta.get('chartPreviousClose', meta.get('previousClose', reg_price)) final_price = pre_price if pre_price else (post_price if post_price else reg_price) change = final_price - prev_close change_pct = (change / prev_close * 100) if prev_close > 0 else 0 result_item = { 'price': round(final_price, 2), 'change': round(change, 2), 'change_pct': round(change_pct, 2), 'volume': meta.get('regularMarketVolume', 0) } kline_data = [] if 'timestamp' in chart_result and 'indicators' in chart_result and 'quote' in chart_result['indicators']: timestamps = chart_result['timestamp'] quotes = chart_result['indicators']['quote'][0] for i in range(len(timestamps)): if quotes['open'][i] is not None and quotes['close'][i] is not None: kline_data.append({ "time": timestamps[i], "open": round(quotes['open'][i], 2), "high": round(quotes['high'][i], 2), "low": round(quotes['low'][i], 2), "close": round(quotes['close'][i], 2) }) result_item['chart'] = kline_data results[sym] = result_item except Exception as inner_e: continue self.send_response(200) self.send_header('Content-type', 'application/json') self.send_header('Access-Control-Allow-Origin', '*') self.send_header('Cache-Control', 'no-store, no-cache, must-revalidate, max-age=0') self.end_headers() self.wfile.write(json.dumps(results).encode('utf-8')) except Exception as e: self.send_response(500) self.send_header('Content-type', 'text/plain') self.send_header('Access-Control-Allow-Origin', '*') self.end_headers() self.wfile.write(f"Server Error: {str(e)}".encode('utf-8'))
